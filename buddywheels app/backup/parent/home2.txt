import { Ionicons } from "@expo/vector-icons";
import { LinearGradient } from "expo-linear-gradient";
import { router } from "expo-router";
import {
  collection,
  doc,
  getDoc,
  getDocs,
  onSnapshot,
  query,
  where,
} from "firebase/firestore";
import React, { useCallback, useEffect, useRef, useState } from "react";
import {
  ActivityIndicator,
  Animated,
  Image,
  RefreshControl,
  ScrollView,
  StatusBar,
  StyleSheet,
  Text,
  TouchableOpacity,
  View,
} from "react-native";
// @ts-ignore
import { auth, db } from "../../firebaseConfig";

interface ChildAttendance {
  studentID: string;
  studentName: string;
  present: number;
  absent: number;
  totalWorkdays: number;
  attendanceRate: number;
}

export default function ParentHome() {
  const [userData, setUserData] = useState<any>(null);
  const [refreshing, setRefreshing] = useState(false);
  const [childrenAttendance, setChildrenAttendance] = useState<
    ChildAttendance[]
  >([]);
  const [loadingStats, setLoadingStats] = useState(true);

  const scaleAnim1 = useRef(new Animated.Value(1)).current;
  const scaleAnim2 = useRef(new Animated.Value(1)).current;
  const fadeAnim = useRef(new Animated.Value(0)).current;

  useEffect(() => {
    Animated.timing(fadeAnim, {
      toValue: 1,
      duration: 800,
      useNativeDriver: true,
    }).start();
  }, []);

  const fetchUserData = useCallback(async () => {
    try {
      // @ts-ignore
      const user = auth.currentUser;
      if (!user) return;

      const userRef = doc(db, "users", user.uid);
      const docSnap = await getDoc(userRef);

      if (docSnap.exists()) {
        setUserData(docSnap.data());
      }
    } catch (error) {
      console.error("Error fetching user data:", error);
    }
  }, []);

  // Calculate workdays from start of month until today
  const calculateWorkdays = (startDate: Date, endDate: Date): number => {
    let workdays = 0;
    const current = new Date(startDate);

    while (current <= endDate) {
      const day = current.getDay();
      // Count only weekdays (Monday=1 to Friday=5)
      if (day !== 0 && day !== 6) {
        workdays++;
      }
      current.setDate(current.getDate() + 1);
    }

    return workdays;
  };

  const fetchAttendanceStats = useCallback(async () => {
    try {
      setLoadingStats(true);
      // @ts-ignore
      const user = auth.currentUser;
      if (!user) return;

      // Get current user's parent ID
      const userRef = doc(db, "users", user.uid);
      const userDoc = await getDoc(userRef);
      if (!userDoc.exists()) return;

      const parentID = userDoc.data().id;

      // Get all children of this parent
      const studentsQuery = query(
        collection(db, "students"),
        where("parentID", "==", parentID)
      );
      const studentsSnapshot = await getDocs(studentsQuery);

      if (studentsSnapshot.empty) {
        setLoadingStats(false);
        return;
      }

      // Get date range from start of month to today
      const today = new Date();
      const startDate = new Date(today.getFullYear(), today.getMonth(), 1);

      const formatDate = (date: Date) => {
        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, "0");
        const day = String(date.getDate()).padStart(2, "0");
        return `${year}-${month}-${day}`;
      };

      const startStr = formatDate(startDate);
      const todayStr = formatDate(today);

      // Calculate total workdays from start of month to today
      const totalWorkdays = calculateWorkdays(startDate, today);

      // Process each child individually
      const childrenData: ChildAttendance[] = [];

      for (const studentDoc of studentsSnapshot.docs) {
        const studentData = studentDoc.data();
        const studentID = studentData.studentID;
        const studentName = studentData.fullName || "Unknown";

        // Fetch attendance records for this specific child
        const attendanceQuery = query(
          collection(db, "attendance"),
          where("studentID", "==", studentID),
          where("date", ">=", startStr),
          where("date", "<=", todayStr)
        );

        const attendanceSnapshot = await getDocs(attendanceQuery);

        // Create a Set of dates with present status (status === 1)
        const presentDates = new Set<string>();

        attendanceSnapshot.forEach((doc) => {
          const data = doc.data();
          const date = new Date(data.date);
          const day = date.getDay();

          // Only count weekdays
          if (day !== 0 && day !== 6) {
            if (data.status === 1) {
              presentDates.add(data.date);
            }
          }
        });

        // Present = days with status 1
        const presentCount = presentDates.size;
        // Absent = total workdays - present days
        const absentCount = totalWorkdays - presentCount;

        const rate =
          totalWorkdays > 0 ? (presentCount / totalWorkdays) * 100 : 0;

        childrenData.push({
          studentID,
          studentName,
          present: presentCount,
          absent: absentCount,
          totalWorkdays,
          attendanceRate: Math.round(rate),
        });
      }

      setChildrenAttendance(childrenData);
      setLoadingStats(false);
    } catch (error) {
      console.error("Error fetching attendance stats:", error);
      setLoadingStats(false);
    }
  }, []);

  useEffect(() => {
    // @ts-ignore
    const user = auth.currentUser;
    if (!user) return;

    const userRef = doc(db, "users", user.uid);
    const unsubscribe = onSnapshot(userRef, (docSnap) => {
      if (docSnap.exists()) {
        setUserData(docSnap.data());
      }
    });

    return () => unsubscribe();
  }, []);

  useEffect(() => {
    fetchAttendanceStats();
  }, [fetchAttendanceStats]);

  const onRefresh = useCallback(async () => {
    setRefreshing(true);
    await fetchUserData();
    await fetchAttendanceStats();
    setTimeout(() => setRefreshing(false), 800);
  }, [fetchUserData, fetchAttendanceStats]);

  const handlePressIn = (anim: Animated.Value) => {
    Animated.spring(anim, {
      toValue: 0.95,
      useNativeDriver: true,
    }).start();
  };

  const handlePressOut = (anim: Animated.Value) => {
    Animated.spring(anim, {
      toValue: 1,
      friction: 3,
      tension: 40,
      useNativeDriver: true,
    }).start();
  };

  // Calculate overall stats for the quick stats cards
  const overallStats = childrenAttendance.reduce(
    (acc, child) => ({
      totalPresent: acc.totalPresent + child.present,
      totalAbsent: acc.totalAbsent + child.absent,
      totalDays: acc.totalDays + child.totalWorkdays,
    }),
    { totalPresent: 0, totalAbsent: 0, totalDays: 0 }
  );

  const overallRate =
    overallStats.totalDays > 0
      ? Math.round((overallStats.totalPresent / overallStats.totalDays) * 100)
      : 0;

  return (
    <>
      <StatusBar
        backgroundColor="#67BA03"
        barStyle="light-content"
        translucent={false}
      />
      <View style={styles.container}>
        <LinearGradient
          colors={["#67BA03", "#5AA002", "#4D8902"]}
          start={{ x: 0, y: 0 }}
          end={{ x: 1, y: 1 }}
          style={styles.header}
        >
          {/* Decorative circles */}
          <View style={styles.circleDecor1} />
          <View style={styles.circleDecor2} />
          <View style={styles.circleDecor3} />

          <View style={styles.headerContent}>
            {/* Avatar with glow effect */}
            <View style={styles.avatarContainer}>
              <View style={styles.avatarGlow} />
              <Image
                source={{
                  uri:
                    userData?.ppic ||
                    "https://cdn-icons-png.flaticon.com/512/3135/3135715.png",
                }}
                style={styles.avatar}
              />
              <View style={styles.onlineIndicator} />
            </View>

            {/* Text content */}
            <View style={styles.headerTextContainer}>
              <Text style={styles.headerSubtitle}>Welcome back,</Text>
              <Text style={styles.headerTitle}>
                {userData?.fullName || "Parent"} 👋
              </Text>
              <View style={styles.roleBadge}>
                <Ionicons name="shield-checkmark" size={12} color="#67BA03" />
                <Text style={styles.roleText}>Parent</Text>
              </View>
            </View>

            {/* Notification with badge */}
            <TouchableOpacity
              activeOpacity={0.7}
              onPress={() => router.push("/(parent)/notification")}
              style={styles.notificationButton}
            >
              <View style={styles.notificationIconContainer}>
                <Ionicons name="notifications" size={24} color="#67BA03" />
                <View style={styles.notificationBadge}>
                  <Text style={styles.notificationBadgeText}>3</Text>
                </View>
              </View>
            </TouchableOpacity>
          </View>
        </LinearGradient>
        <ScrollView
          showsVerticalScrollIndicator={false}
          refreshControl={
            <RefreshControl
              refreshing={refreshing}
              onRefresh={onRefresh}
              tintColor="#67BA03"
              colors={["#67BA03"]}
            />
          }
        >
          {/* Welcome Card with gradient border */}
          <Animated.View style={[styles.welcomeCard, { opacity: fadeAnim }]}>
            <View style={styles.welcomeIconContainer}>
              <LinearGradient
                colors={["#67BA03", "#5AA002"]}
                style={styles.welcomeIconBg}
              >
                <Ionicons name="home" size={24} color="#fff" />
              </LinearGradient>
            </View>
            <Text style={styles.welcomeTitle}>Parent Dashboard</Text>
            <Text style={styles.welcomeText}>
              Manage, monitor, and stay updated with your child's daily commute
              and attendance effortlessly.
            </Text>
          </Animated.View>

          {/* Monthly Attendance Analytics Card - For Each Child */}
          <TouchableOpacity
            activeOpacity={0.8}
            onPress={() => router.push("/childattendance")}
            style={{ marginHorizontal: 20, marginTop: 16 }}
          >
            <Animated.View
              style={[styles.analyticsCard, { opacity: fadeAnim }]}
            >
              <View style={styles.analyticsHeader}>
                <View style={styles.analyticsIconBg}>
                  <Ionicons name="bar-chart" size={24} color="#2196F3" />
                </View>
                <View style={{ flex: 1 }}>
                  <Text style={styles.analyticsTitle}>Monthly Attendance</Text>
                  <Text style={styles.analyticsSubtitle}>
                    {new Date().toLocaleDateString("en-MY", {
                      month: "long",
                      year: "numeric",
                    })}{" "}
                    • As of{" "}
                    {new Date().toLocaleDateString("en-MY", {
                      day: "numeric",
                      month: "short",
                    })}
                  </Text>
                </View>
                <Ionicons name="chevron-forward" size={20} color="#999" />
              </View>

              {loadingStats ? (
                <View style={styles.analyticsLoading}>
                  <ActivityIndicator size="small" color="#2196F3" />
                </View>
              ) : (
                <>
                  {/* Individual Child Cards */}
                  {childrenAttendance.map((child, index) => (
                    <View key={child.studentID}>
                      <View style={styles.childCard}>
                        <View style={styles.childHeader}>
                          <View style={styles.childIconBg}>
                            <Ionicons name="person" size={20} color="#2196F3" />
                          </View>
                          <Text style={styles.childName}>
                            {child.studentName}
                          </Text>
                        </View>

                        <View style={styles.childStats}>
                          <View style={styles.childStatItem}>
                            <Text style={styles.childStatValue}>
                              {child.present}/{child.totalWorkdays}
                            </Text>
                            <Text style={styles.childStatLabel}>Present</Text>
                          </View>

                          <View style={styles.childStatDivider} />

                          <View style={styles.childStatItem}>
                            <Text
                              style={[
                                styles.childStatValue,
                                { color: "#E53935" },
                              ]}
                            >
                              {child.absent}/{child.totalWorkdays}
                            </Text>
                            <Text style={styles.childStatLabel}>Absent</Text>
                          </View>

                          <View style={styles.childStatDivider} />

                          <View style={styles.childStatItem}>
                            <Text
                              style={[
                                styles.childStatValue,
                                { color: "#FF9800" },
                              ]}
                            >
                              {child.attendanceRate}%
                            </Text>
                            <Text style={styles.childStatLabel}>Rate</Text>
                          </View>
                        </View>
                      </View>

                      {index < childrenAttendance.length - 1 && (
                        <View style={styles.childDivider} />
                      )}
                    </View>
                  ))}

                  {childrenAttendance.length === 0 && (
                    <View style={styles.noDataContainer}>
                      <Ionicons name="school-outline" size={48} color="#ccc" />
                      <Text style={styles.noDataText}>
                        No children registered
                      </Text>
                    </View>
                  )}

                  <View style={styles.analyticsFooter}>
                    <Text style={styles.analyticsFooterText}>
                      Tap to view detailed analytics
                    </Text>
                  </View>
                </>
              )}
            </Animated.View>
          </TouchableOpacity>

          {/* Quick Stats Cards */}
          <View style={styles.statsContainer}>
            <View style={styles.statCard}>
              <View style={[styles.statIconBg, { backgroundColor: "#E8F5E9" }]}>
                <Ionicons name="people" size={20} color="#67BA03" />
              </View>
              <Text style={styles.statValue}>{childrenAttendance.length}</Text>
              <Text style={styles.statLabel}>Children</Text>
            </View>
            <View style={styles.statCard}>
              <View style={[styles.statIconBg, { backgroundColor: "#FFF3E0" }]}>
                <Ionicons name="calendar" size={20} color="#FF9800" />
              </View>
              <Text style={styles.statValue}>{overallRate}%</Text>
              <Text style={styles.statLabel}>Attendance</Text>
            </View>
            <View style={styles.statCard}>
              <View style={[styles.statIconBg, { backgroundColor: "#E3F2FD" }]}>
                <Ionicons name="checkmark-circle" size={20} color="#2196F3" />
              </View>
              <Text style={styles.statValue}>{overallStats.totalPresent}</Text>
              <Text style={styles.statLabel}>Present</Text>
            </View>
          </View>

          {/* Main Action Cards */}
          <View style={styles.sectionHeader}>
            <Text style={styles.sectionTitle}>Quick Actions</Text>
            <View style={styles.sectionLine} />
          </View>

          <View style={styles.buttonContainer}>
            {/* Child Dashboard Card */}
            <Animated.View style={{ transform: [{ scale: scaleAnim1 }] }}>
              <TouchableOpacity
                activeOpacity={0.9}
                onPressIn={() => handlePressIn(scaleAnim1)}
                onPressOut={() => handlePressOut(scaleAnim1)}
                onPress={() => router.push("/childdashboard")}
                style={styles.actionCard}
              >
                <LinearGradient
                  colors={["#67BA03", "#5AA002"]}
                  start={{ x: 0, y: 0 }}
                  end={{ x: 1, y: 1 }}
                  style={styles.actionCardGradient}
                >
                  <View style={styles.actionCardPattern} />
                  <View style={styles.actionCardContent}>
                    <View style={styles.actionIconContainer}>
                      <View style={styles.actionIconBg}>
                        <Ionicons
                          name="people-circle"
                          size={32}
                          color="#67BA03"
                        />
                      </View>
                    </View>
                    <View style={styles.actionTextContainer}>
                      <Text style={styles.actionTitle}>Child Dashboard</Text>
                      <Text style={styles.actionSubtitle}>
                        Add and manage your children's rides
                      </Text>
                    </View>
                    <View style={styles.actionArrow}>
                      <Ionicons name="arrow-forward" size={20} color="#fff" />
                    </View>
                  </View>
                </LinearGradient>
              </TouchableOpacity>
            </Animated.View>

            {/* Child Attendance Card */}
            <Animated.View style={{ transform: [{ scale: scaleAnim2 }] }}>
              <TouchableOpacity
                activeOpacity={0.9}
                onPressIn={() => handlePressIn(scaleAnim2)}
                onPressOut={() => handlePressOut(scaleAnim2)}
                onPress={() => router.push("/childattendance")}
                style={styles.actionCard}
              >
                <LinearGradient
                  colors={["#2196F3", "#1976D2"]}
                  start={{ x: 0, y: 0 }}
                  end={{ x: 1, y: 1 }}
                  style={styles.actionCardGradient}
                >
                  <View style={styles.actionCardPattern} />
                  <View style={styles.actionCardContent}>
                    <View style={styles.actionIconContainer}>
                      <View style={styles.actionIconBg}>
                        <Ionicons name="calendar" size={32} color="#2196F3" />
                      </View>
                    </View>
                    <View style={styles.actionTextContainer}>
                      <Text style={styles.actionTitle}>Child Attendance</Text>
                      <Text style={styles.actionSubtitle}>
                        Track attendance records and excuses
                      </Text>
                    </View>
                    <View style={styles.actionArrow}>
                      <Ionicons name="arrow-forward" size={20} color="#fff" />
                    </View>
                  </View>
                </LinearGradient>
              </TouchableOpacity>
            </Animated.View>
          </View>

          <View style={{ height: 30 }} />
        </ScrollView>
      </View>
    </>
  );
}
