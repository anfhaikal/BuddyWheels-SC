type Student = {
  id: string;
  studentID: string;
  name: string;
  status: "Present" | "Absent";
  excuseStatus?: "Pending" | "Approved" | "Denied";
};

type ClassInfo = {
  grade: number;
  className: string;
  teacherName: string;
};

type ExcuseLetter = {
  id?: string;
  studentID?: string;
  date?: string;
  file?: string; // base64
  fileName?: string;
  name?: string;
  reason?: string;
  status?: string;
  time?: string;
  teacherComment?: string;

        const data = doc.data();

        if (data.studentID) {
          excuseMap[data.studentID] = data.status || "Pending";
        }
      });

      setExcuseLetters(excuseMap);
    });

      where("classID", "==", classID)
    );

      if (!classSnapshot.empty) {
        const classData = classSnapshot.docs[0].data();

        // Clean up previous teacher listener if it exists

          teacherUnsubscribe();
          teacherUnsubscribe = null;

          const teacherQuery = query(
            collection(db, "Teacher"),

            if (!teacherSnapshot.empty) {

          });

            grade: classData.grade,
            className: classData.className,
            teacherName,
          });
        }
      }
    });

    // Cleanup function
    return () => {
      unsubscribeClass();
      if (teacherUnsubscribe) {
        teacherUnsubscribe();

      where("date", "==", dateStr),

    // Listener for students
    const unsubscribeStudents = onSnapshot(studentQuery, (studentSnap) => {
      const studentList = studentSnap.docs.map((docItem) => ({
        id: docItem.id,
        ...docItem.data(),
      })) as { id: string; name: string; studentID: string }[];

      // Listener for attendance (nested to have access to studentList)
      const unsubscribeAttendance = onSnapshot(
        attendanceQuery,
        (attendanceSnap) => {
          const attendanceMap: Record<string, "Present" | "Absent"> = {};
          attendanceSnap.forEach((docItem) => {
            const data = docItem.data();
            const status = data.status === 1 ? "Present" : "Absent";

          });

            id: student.id,
            studentID: student.studentID,
            name: student.name,
            status: attendanceMap[student.studentID] || "Absent",
            excuseStatus: excuseLetters[student.studentID] as
              | "Pending"
              | "Approved"
              | "Denied"
              | undefined,
          }));

          setStudents(combined);
        }
      );

      // Store unsubscribeAttendance for cleanup
      return () => unsubscribeAttendance();
    });

    // Cleanup function
    return () => {
      unsubscribeStudents();
    };

            collection(db, "Teacher"),

          const teacherSnapshot = await getDocs(teacherQuery);
          if (!teacherSnapshot.empty) {
            const teacherData = teacherSnapshot.docs[0].data();
            teacherName = teacherData.fullName || "Unknown Teacher";

        }

        setClassInfo({
          grade: classData.grade,
          className: classData.className,
          teacherName,
        });

    } catch (error) {
      console.error("❌ Error fetching class info:", error);

  const handleStatusChange = (id: string, status: "Present" | "Absent") => {
    const updated = students.map((s) => (s.id === id ? { ...s, status } : s));
    setStudents(updated);
  };

  const toggleEdit = async () => {
    if (isEditing) {
      try {
        const dateStr = getMalaysiaDateString(selectedDate);
        const notificationsToSend: Array<{
          studentID: string;
          name: string;
          previousStatus: string;
          newStatus: string;
        }> = [];

        // Fetch existing attendance to detect changes
        const attendanceQuery = query(
          collection(db, "attendance"),
          where("date", "==", dateStr),
          where("classID", "==", classID)
        );
        const existingAttendance = await getDocs(attendanceQuery);
        const existingMap: Record<string, number> = {};

        existingAttendance.forEach((doc) => {
          const data = doc.data();
          existingMap[data.studentID] = data.status;
        });

        // Update attendance records and track changes
        for (const student of students) {
          const attendanceRef = doc(
            db,
            "attendance",
            `${dateStr}_${student.studentID}`
          );

          const newStatus = student.status === "Present" ? 1 : 0;
          const oldStatus = existingMap[student.studentID]; // Can be undefined, 0, or 1

          await setDoc(attendanceRef, {
            studentID: student.studentID,
            name: student.name,
            classID: classID,
            date: dateStr,
            status: newStatus,
          });

          // ✅ Fixed: Check if oldStatus is undefined (no record) or 0 (Absent)
          // and newStatus is 1 (Present)
          if ((oldStatus === undefined || oldStatus === 0) && newStatus === 1) {
            notificationsToSend.push({
              studentID: student.studentID,
              name: student.name,
              previousStatus: "Absent",
              newStatus: "Present",
            });
          }
        }

        // Send notifications to parents only for status changes
        if (notificationsToSend.length > 0) {
          await sendAttendanceNotifications(notificationsToSend, dateStr);
        }

        console.log("✅ Attendance successfully updated in Firestore");
        alert("Attendance successfully updated!");
      } catch (error) {
        console.error("❌ Error updating attendance:", error);
        alert("Error updating attendance");
      }
    }

    setIsEditing((prev) => !prev);
  };

  const sendAttendanceNotifications = async (
    studentsPresent: Array<{
      studentID: string;
      name: string;
      previousStatus: string;
      newStatus: string;
    }>,
    dateStr: string
  ) => {
    try {
      for (const student of studentsPresent) {
        // Step 1: Get parentID from student
        const studentQuery = query(
          collection(db, "students"),
          where("studentID", "==", student.studentID)
        );
        const studentSnapshot = await getDocs(studentQuery);

        if (!studentSnapshot.empty) {
          const studentData = studentSnapshot.docs[0].data();
          const parentID = studentData.parentID;

          if (parentID) {
            // Step 2: Verify Parent exists (optional but good for debugging)
            const parentQuery = query(
              collection(db, "Parent"),
              where("id", "==", parentID)
            );
            const parentSnapshot = await getDocs(parentQuery);

            if (!parentSnapshot.empty) {
              // Step 3: Get auth UID from users collection
              const userQuery = query(
                collection(db, "users"),
                where("id", "==", parentID)
              );
              const userSnapshot = await getDocs(userQuery);

              if (!userSnapshot.empty) {
                const authUID = userSnapshot.docs[0].id; // This is the auth UID!

                // Step 4: Create notification with notiType
                await addDoc(collection(db, "notifications"), {
                  userId: authUID, // ✅ Correct auth UID
                  title: "✅ Attendance Update",
                  message: `Good news! ${
                    student.name
                  } has been marked as Present for ${new Date(
                    dateStr
                  ).toLocaleDateString("en-MY", {
                    weekday: "short",
                    year: "numeric",
                    month: "short",
                    day: "numeric",
                  })}.`,
                  type: "success",
                  notiType: "attendance", // ✅ Added notiType attribute
                  isRead: false,
                  timestamp: Timestamp.now(),
                });

                console.log(
                  `✅ Notification sent to parent of ${student.name}`
                );
              }
            }
          }
        }
      }
    } catch (error) {
      console.error("❌ Error sending notifications:", error);
    }

    const excuseRef = doc(db, "excuseLetter", docId);

      if (snap.exists()) {

        setExcuseLetter({
          id: docId,
          studentID: data.studentID,
          date: data.date,
          file: data.file,
          fileName: data.fileName,
          name: data.name,
          reason: data.reason,
          status: data.status,

      useNativeDriver: true,

  // Approve / Deny handlers: set status and teacherComment in Firestore (merge)
  const updateExcuseStatus = async (newStatus: "Approved" | "Denied") => {
    if (!excuseLetter?.id) {
      Alert.alert("No submission", "No excuse letter to update.");
      return;
    }

    try {
      const docRef = doc(db, "excuseLetter", excuseLetter.id);
      await setDoc(
        docRef,
        {
          status: newStatus,
          teacherComment: comment || "",
        },
        { merge: true }
      );

      // update local state
      setExcuseLetter((prev) =>
        prev ? { ...prev, status: newStatus, teacherComment: comment } : prev
      );

      Alert.alert("Success", `Submission ${newStatus.toLowerCase()}.`);
    } catch (err) {
      console.error("❌ Error updating status:", err);
      Alert.alert("Error", "Failed to update submission status.");
    }
  };

  const formatDate = (date: Date) => {
    return date.toLocaleDateString("en-MY", {
      weekday: "short",
      year: "numeric",
      month: "short",
      day: "numeric",
      timeZone: "Asia/Kuala_Lumpur",

  const getMalaysiaDateString = (date: Date) => {
    const options: Intl.DateTimeFormatOptions = {
      timeZone: "Asia/Kuala_Lumpur",
      year: "numeric",
      month: "2-digit",
      day: "2-digit",
    };

    const formatter = new Intl.DateTimeFormat("en-CA", options);
    const parts = formatter.formatToParts(date);

    const year = parts.find((p) => p.type === "year")?.value || "";
    const month = parts.find((p) => p.type === "month")?.value || "";
    const day = parts.find((p) => p.type === "day")?.value || "";

    return `${year}-${month}-${day}`;
  };

          style={[styles.pageButton, page === 1 && styles.disabledPage]}
        >
          <Ionicons name="chevron-back" size={20} color="#fff" />
        </TouchableOpacity>
        <Text style={styles.pageText}>
          Page {page} / {totalPages}
        </Text>
        <TouchableOpacity
          disabled={page === totalPages}
          onPress={() => setPage(page + 1)}
          style={[
            styles.pageButton,
            page === totalPages && styles.disabledPage,
          ]}
        >
          <Ionicons name="chevron-forward" size={20} color="#fff" />
        </TouchableOpacity>
      </View>

      {/* Overlay */}
      {selectedStudent && (
        <Animated.View
          style={[
            styles.overlayContainer,
            { transform: [{ translateY: slideAnim }] },
          ]}
        >
          <TouchableWithoutFeedback onPress={Keyboard.dismiss}>
            <View style={{ flex: 1 }} />
          </TouchableWithoutFeedback>

          <KeyboardAvoidingView
            behavior={Platform.OS === "ios" ? "padding" : "height"}
            style={{ maxHeight: "80%" }}
          >
            <ScrollView
              style={styles.overlayContent}
              keyboardShouldPersistTaps="handled"
            >
              <Text style={styles.overlayName}>{selectedStudent.name}</Text>
              <Text style={styles.overlayDate}>{formatDate(selectedDate)}</Text>
              <Text style={styles.excuseHeader}>Excuse Letter</Text>

              <View style={styles.excuseTable}>
                <View style={styles.excuseRow}>
                  <Text style={styles.excuseLabel}>Submission Review</Text>
                  <Text
                    style={[
                      styles.excuseValue,
                      excuseLetter?.status === "Pending" && {
                        color: "#FF9800",
                        fontWeight: "bold",
                      },
                      excuseLetter?.status === "Approved" && {
                        color: "#2E7D32",
                        fontWeight: "bold",
                      },
                      excuseLetter?.status === "Denied" && {
                        color: "#E53935",
                        fontWeight: "bold",
                      },
                    ]}
                  >
                    {excuseLetter?.status || "No submission yet"}
                  </Text>
                </View>

                <View style={styles.excuseRow}>
                  <Text style={styles.excuseLabel}>Submitted On</Text>
                  <Text style={styles.excuseValue}>
                    {/* keep previous behaviour — not modified */}
                    {/* If you want the computed remaining time show here, we could reuse parent's computePendingTimeText */}
                    {excuseLetter?.time
                      ? new Date(excuseLetter.time).toLocaleString()
                      : "—"}
                  </Text>
                </View>

                <View style={styles.excuseRow}>
                  <Text style={styles.excuseLabel}>File Submission</Text>
                  <View style={{ flex: 2 }}>
                    {excuseLetter?.fileName ? (
                      <>
                        <TouchableOpacity onPress={openPdfInApp}>
                          <Text
                            style={{ color: "#67BA03", fontWeight: "bold" }}
                          >
                            {excuseLetter.fileName}
                          </Text>
                        </TouchableOpacity>
                        <Text
                          style={{ fontSize: 12, color: "#777", marginTop: 4 }}
                        >
                          Tap to open inside app
                        </Text>
                      </>
                    ) : (
                      <Text style={styles.excuseValue}>No file</Text>
                    )}
                  </View>
                </View>

                <View style={styles.excuseRow}>
                  <Text style={styles.excuseLabel}>Comments</Text>
                  <TextInput
                    style={styles.commentInput}
                    placeholder="Add teacher comments..."
                    placeholderTextColor="#b9b3b3ff"
                    value={comment}
                    onChangeText={setComment}
                    multiline
                  />
                </View>
              </View>

              {/* In-app PDF viewer area */}
              {showPdf && excuseLetter?.file && (
                <View style={{ height: 420, marginBottom: 12 }}>
                  <View
                    style={{
                      flexDirection: "row",
                      justifyContent: "space-between",
                      alignItems: "center",
                      marginBottom: 6,
                    }}
                  >
                    <Text style={{ fontWeight: "600" }}>
                      {excuseLetter.fileName}
                    </Text>
                    <TouchableOpacity onPress={() => setShowPdf(false)}>
                      <Text style={{ color: "#E53935", fontWeight: "bold" }}>
                        Close PDF
                      </Text>
                    </TouchableOpacity>
                  </View>
                  <WebView
                    originWhitelist={["*"]}
                    source={{
                      uri: `data:application/pdf;base64,${excuseLetter.file}`,
                    }}
                    style={{ flex: 1, borderRadius: 8, overflow: "hidden" }}
                    // On some Android devices large base64 PDFs might be problematic; in that case consider uploading to storage and loading via URL
                  />
                </View>
              )}

              <View style={styles.overlayButtons}>
                <TouchableOpacity
                  style={styles.approveBtn}
                  onPress={() =>
                    Alert.alert("Approve", "Approve this submission?", [
                      { text: "Cancel", style: "cancel" },
                      {
                        text: "Approve",
                        onPress: () => updateExcuseStatus("Approved"),
                      },
                    ])
                  }
                >
                  <Text style={styles.btnActionText}>APPROVE</Text>
                </TouchableOpacity>
                <TouchableOpacity
                  style={styles.denyBtn}
                  onPress={() =>
                    Alert.alert("Deny", "Deny this submission?", [
                      { text: "Cancel", style: "cancel" },
                      {
                        text: "Deny",
                        onPress: () => updateExcuseStatus("Denied"),
                      },
                    ])
                  }
                >
                  <Text style={styles.btnActionText}>DENY</Text>
                </TouchableOpacity>
              </View>

              <TouchableOpacity onPress={closeOverlay} style={styles.closeBtn}>
                <Text style={styles.closeBtnText}>Close</Text>
              </TouchableOpacity>
            </ScrollView>
          </KeyboardAvoidingView>
        </Animated.View>
      )}
    </SafeAreaView>
  );
}
