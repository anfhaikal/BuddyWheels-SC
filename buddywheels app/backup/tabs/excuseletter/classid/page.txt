// class.tsx (teacher page) — improved UI with consistent green theme and Mark All feature
import { Ionicons } from "@expo/vector-icons";
import DateTimePicker from "@react-native-community/datetimepicker";
import { useFocusEffect } from "@react-navigation/native";
import { LinearGradient } from "expo-linear-gradient";
import { router, useLocalSearchParams } from "expo-router";
import {
  addDoc,
  collection,
  doc,
  getDocs,
  onSnapshot,
  query,
  setDoc,
  Timestamp,
  where,
} from "firebase/firestore";
import React, { useEffect, useRef, useState } from "react";
import {
  Alert,
  Animated,
  Dimensions,
  Keyboard,
  KeyboardAvoidingView,
  Platform,
  ScrollView,
  StatusBar,
  StyleSheet,
  Text,
  TextInput,
  TouchableOpacity,
  TouchableWithoutFeedback,
  View,
} from "react-native";
import { SafeAreaView } from "react-native-safe-area-context";
import { WebView } from "react-native-webview";
import { db } from "../../../firebaseConfig";

type Student = {
  id: string;
  studentID: string;
  name: string;
  status: "Present" | "Absent";
  excuseStatus?: "Pending" | "Approved" | "Denied";
};

type ClassInfo = {
  grade: number;
  className: string;
  teacherName: string;
};

type ExcuseLetter = {
  id?: string;
  studentID?: string;
  date?: string;
  file?: string;
  fileName?: string;
  name?: string;
  reason?: string;
  status?: string;
  time?: string;
  teacherComment?: string;
};

export default function ClassAttendanceScreen() {
  const { classID } = useLocalSearchParams<{ classID: string }>();
  const [students, setStudents] = useState<Student[]>([]);
  const [classInfo, setClassInfo] = useState<ClassInfo | null>(null);
  const [isEditing, setIsEditing] = useState(false);
  const [page, setPage] = useState(1);
  const [selectedStudent, setSelectedStudent] = useState<Student | null>(null);
  const [comment, setComment] = useState("");
  const [selectedDate, setSelectedDate] = useState(new Date());
  const [showDatePicker, setShowDatePicker] = useState(false);
  const [excuseUnsubscribe, setExcuseUnsubscribe] = useState<
    (() => void) | null
  >(null);
  const [excuseLetters, setExcuseLetters] = useState<Record<string, string>>(
    {}
  );
  const [excuseLetter, setExcuseLetter] = useState<ExcuseLetter | null>(null);
  const [showPdf, setShowPdf] = useState(false);

  const slideAnim = useRef(
    new Animated.Value(Dimensions.get("window").height)
  ).current;

  const studentsPerPage = 10;
  const totalPages = Math.max(1, Math.ceil(students.length / studentsPerPage));
  const startIdx = (page - 1) * studentsPerPage;
  const currentStudents = students.slice(startIdx, startIdx + studentsPerPage);

  const presentCount = students.filter((s) => s.status === "Present").length;
  const absentCount = students.filter((s) => s.status === "Absent").length;

  useEffect(() => {
    if (!classID) return;

    const dateStr = getMalaysiaDateString(selectedDate);

    const excuseQuery = query(
      collection(db, "excuseLetter"),
      where("date", "==", dateStr)
    );

    const unsubscribe = onSnapshot(excuseQuery, (snapshot) => {
      const excuseMap: Record<string, string> = {};

      snapshot.forEach((doc) => {
        const data = doc.data();
        if (data.studentID) {
          excuseMap[data.studentID] = data.status || "Pending";
        }
      });

      setExcuseLetters(excuseMap);
    });

    return () => unsubscribe();
  }, [selectedDate, classID]);

  useFocusEffect(
    React.useCallback(() => {
      const now = new Date();
      setSelectedDate(now);
    }, [])
  );

  useEffect(() => {
    if (!classID) return;

    let teacherUnsubscribe: (() => void) | null = null;

    const classQuery = query(
      collection(db, "classes"),
      where("classID", "==", classID)
    );

    const unsubscribeClass = onSnapshot(classQuery, (classSnapshot) => {
      if (!classSnapshot.empty) {
        const classData = classSnapshot.docs[0].data();

        if (teacherUnsubscribe) {
          teacherUnsubscribe();
          teacherUnsubscribe = null;
        }

        let teacherName = "No teacher assigned";

        if (classData.teacherID) {
          const teacherQuery = query(
            collection(db, "Teacher"),
            where("id", "==", classData.teacherID)
          );

          teacherUnsubscribe = onSnapshot(teacherQuery, (teacherSnapshot) => {
            if (!teacherSnapshot.empty) {
              const teacherData = teacherSnapshot.docs[0].data();
              teacherName = teacherData.fullName || "Unknown Teacher";
            } else {
              teacherName = "Unknown Teacher";
            }

            setClassInfo({
              grade: classData.grade,
              className: classData.className,
              teacherName,
            });
          });
        } else {
          setClassInfo({
            grade: classData.grade,
            className: classData.className,
            teacherName,
          });
        }
      }
    });

    return () => {
      unsubscribeClass();
      if (teacherUnsubscribe) {
        teacherUnsubscribe();
      }
    };
  }, [classID]);

  useEffect(() => {
    if (!classID) return;

    const dateStr = getMalaysiaDateString(selectedDate);

    const studentQuery = query(
      collection(db, "students"),
      where("classID", "==", classID)
    );

    const attendanceQuery = query(
      collection(db, "attendance"),
      where("date", "==", dateStr),
      where("classID", "==", classID)
    );

    const unsubscribeStudents = onSnapshot(studentQuery, (studentSnap) => {
      const studentList = studentSnap.docs.map((docItem) => ({
        id: docItem.id,
        ...docItem.data(),
      })) as { id: string; name: string; studentID: string }[];

      const unsubscribeAttendance = onSnapshot(
        attendanceQuery,
        (attendanceSnap) => {
          const attendanceMap: Record<string, "Present" | "Absent"> = {};
          attendanceSnap.forEach((docItem) => {
            const data = docItem.data();
            const status = data.status === 1 ? "Present" : "Absent";
            attendanceMap[data.studentID] = status;
          });

          const combined = studentList.map((student) => ({
            id: student.id,
            studentID: student.studentID,
            name: student.name,
            status: attendanceMap[student.studentID] || "Absent",
            excuseStatus: excuseLetters[student.studentID] as
              | "Pending"
              | "Approved"
              | "Denied"
              | undefined,
          }));

          setStudents(combined);
        }
      );

      return () => unsubscribeAttendance();
    });

    return () => {
      unsubscribeStudents();
    };
  }, [selectedDate, classID, excuseLetters]);

  const handleStatusChange = (id: string, status: "Present" | "Absent") => {
    const updated = students.map((s) => (s.id === id ? { ...s, status } : s));
    setStudents(updated);
  };

  const markAllStudents = (status: "Present" | "Absent") => {
    const updated = students.map((s) => ({ ...s, status }));
    setStudents(updated);
  };

  const toggleEdit = async () => {
    if (isEditing) {
      try {
        const dateStr = getMalaysiaDateString(selectedDate);
        const notificationsToSend: Array<{
          studentID: string;
          name: string;
          previousStatus: string;
          newStatus: string;
        }> = [];

        const attendanceQuery = query(
          collection(db, "attendance"),
          where("date", "==", dateStr),
          where("classID", "==", classID)
        );
        const existingAttendance = await getDocs(attendanceQuery);
        const existingMap: Record<string, number> = {};

        existingAttendance.forEach((doc) => {
          const data = doc.data();
          existingMap[data.studentID] = data.status;
        });

        for (const student of students) {
          const attendanceRef = doc(
            db,
            "attendance",
            `${dateStr}_${student.studentID}`
          );

          const newStatus = student.status === "Present" ? 1 : 0;
          const oldStatus = existingMap[student.studentID];

          await setDoc(attendanceRef, {
            studentID: student.studentID,
            name: student.name,
            classID: classID,
            date: dateStr,
            status: newStatus,
          });

          if ((oldStatus === undefined || oldStatus === 0) && newStatus === 1) {
            notificationsToSend.push({
              studentID: student.studentID,
              name: student.name,
              previousStatus: "Absent",
              newStatus: "Present",
            });
          }
        }

        if (notificationsToSend.length > 0) {
          await sendAttendanceNotifications(notificationsToSend, dateStr);
        }

        console.log("✅ Attendance successfully updated in Firestore");
        alert("Attendance successfully updated!");
      } catch (error) {
        console.error("❌ Error updating attendance:", error);
        alert("Error updating attendance");
      }
    }

    setIsEditing((prev) => !prev);
  };

  const sendAttendanceNotifications = async (
    studentsPresent: Array<{
      studentID: string;
      name: string;
      previousStatus: string;
      newStatus: string;
    }>,
    dateStr: string
  ) => {
    try {
      for (const student of studentsPresent) {
        const studentQuery = query(
          collection(db, "students"),
          where("studentID", "==", student.studentID)
        );
        const studentSnapshot = await getDocs(studentQuery);

        if (!studentSnapshot.empty) {
          const studentData = studentSnapshot.docs[0].data();
          const parentID = studentData.parentID;

          if (parentID) {
            const parentQuery = query(
              collection(db, "Parent"),
              where("id", "==", parentID)
            );
            const parentSnapshot = await getDocs(parentQuery);

            if (!parentSnapshot.empty) {
              const userQuery = query(
                collection(db, "users"),
                where("id", "==", parentID)
              );
              const userSnapshot = await getDocs(userQuery);

              if (!userSnapshot.empty) {
                const authUID = userSnapshot.docs[0].id;

                await addDoc(collection(db, "notifications"), {
                  userId: authUID,
                  title: "✅ Attendance Update",
                  message: `Good news! ${
                    student.name
                  } has been marked as Present for ${new Date(
                    dateStr
                  ).toLocaleDateString("en-MY", {
                    weekday: "short",
                    year: "numeric",
                    month: "short",
                    day: "numeric",
                  })}.`,
                  type: "success",
                  notiType: "attendance",
                  isRead: false,
                  timestamp: Timestamp.now(),
                });

                console.log(
                  `✅ Notification sent to parent of ${student.name}`
                );
              }
            }
          }
        }
      }
    } catch (error) {
      console.error("❌ Error sending notifications:", error);
    }
  };

  const changeDate = (days: number) => {
    const newDate = new Date(selectedDate);
    newDate.setDate(newDate.getDate() + days);
    setSelectedDate(newDate);
  };

  const onDateChange = (event: any, date?: Date) => {
    setShowDatePicker(Platform.OS === "ios");
    if (date) {
      setSelectedDate(date);
    }
  };

  const openOverlay = async (student: Student) => {
    setSelectedStudent(student);
    setComment("");
    setShowPdf(false);
    setExcuseLetter(null);

    const dateStr = getMalaysiaDateString(selectedDate);
    const docId = `${dateStr}_${student.studentID}`;

    const excuseRef = doc(db, "excuseLetter", docId);
    const unsubscribe = onSnapshot(excuseRef, (snap) => {
      if (snap.exists()) {
        const data = snap.data();
        setExcuseLetter({
          id: docId,
          studentID: data.studentID,
          date: data.date,
          file: data.file,
          fileName: data.fileName,
          name: data.name,
          reason: data.reason,
          status: data.status,
          time: data.time,
          teacherComment: data.teacherComment,
        });
        setComment(data.teacherComment || "");
      } else {
        setExcuseLetter(null);
      }
    });

    setExcuseUnsubscribe(() => unsubscribe);

    Animated.timing(slideAnim, {
      toValue: 0,
      duration: 300,
      useNativeDriver: true,
    }).start();
  };

  const closeOverlay = () => {
    if (excuseUnsubscribe) {
      excuseUnsubscribe();
      setExcuseUnsubscribe(null);
    }

    Animated.timing(slideAnim, {
      toValue: Dimensions.get("window").height,
      duration: 300,
      useNativeDriver: true,
    }).start(() => {
      setSelectedStudent(null);
      setExcuseLetter(null);
      setShowPdf(false);
      setComment("");
    });
  };

  const openPdfInApp = () => {
    if (!excuseLetter?.file) {
      Alert.alert("No file", "No PDF file found for this submission.");
      return;
    }
    setShowPdf(true);
  };

  const updateExcuseStatus = async (newStatus: "Approved" | "Denied") => {
    if (!excuseLetter?.id) {
      Alert.alert("No submission", "No excuse letter to update.");
      return;
    }

    try {
      const docRef = doc(db, "excuseLetter", excuseLetter.id);
      await setDoc(
        docRef,
        {
          status: newStatus,
          teacherComment: comment || "",
        },
        { merge: true }
      );

      setExcuseLetter((prev) =>
        prev ? { ...prev, status: newStatus, teacherComment: comment } : prev
      );

      Alert.alert("Success", `Submission ${newStatus.toLowerCase()}.`);
    } catch (err) {
      console.error("❌ Error updating status:", err);
      Alert.alert("Error", "Failed to update submission status.");
    }
  };

  const formatDate = (date: Date) => {
    return date.toLocaleDateString("en-MY", {
      weekday: "short",
      year: "numeric",
      month: "short",
      day: "numeric",
      timeZone: "Asia/Kuala_Lumpur",
    });
  };

  const getMalaysiaDateString = (date: Date) => {
    const options: Intl.DateTimeFormatOptions = {
      timeZone: "Asia/Kuala_Lumpur",
      year: "numeric",
      month: "2-digit",
      day: "2-digit",
    };

    const formatter = new Intl.DateTimeFormat("en-CA", options);
    const parts = formatter.formatToParts(date);

    const year = parts.find((p) => p.type === "year")?.value || "";
    const month = parts.find((p) => p.type === "month")?.value || "";
    const day = parts.find((p) => p.type === "day")?.value || "";

    return `${year}-${month}-${day}`;
  };

  const getExcuseStatusColor = (status?: string) => {
    switch (status) {
      case "Pending":
        return "#FF9800";
      case "Approved":
        return "#4CAF50";
      case "Denied":
        return "#E53935";
      default:
        return "transparent";
    }
  };

  return (
    <>
      <StatusBar
        backgroundColor="#67BA03"
        barStyle="light-content"
        translucent={false}
      />
      <SafeAreaView style={styles.container} edges={["top"]}>
        {/* Header */}
        <View style={styles.header}>
          <TouchableOpacity onPress={() => router.push("/home")}>
            <Ionicons name="arrow-back" size={24} color="#fff" />
          </TouchableOpacity>
          <Text style={styles.title}>
            {classInfo
              ? `${classInfo.grade} ${classInfo.className}`
              : "Loading..."}
          </Text>
          <View style={{ width: 24 }} />
        </View>

        {/* Class Info with Gradient */}
        <LinearGradient
          colors={["#67BA03", "#5AA002"]}
          start={{ x: 0, y: 0 }}
          end={{ x: 1, y: 1 }}
          style={styles.classInfoBackground}
        >
          <View style={styles.decorativeCircle1} />
          <View style={styles.decorativeCircle2} />

          <View style={styles.classInfoOverlay}>
            <View style={styles.teacherBadge}>
              <Ionicons name="person-circle" size={16} color="#67BA03" />
              <Text
                style={styles.teacher}
                numberOfLines={1}
                ellipsizeMode="tail"
              >
                {classInfo?.teacherName || "Loading..."}
              </Text>
            </View>

            {/* Date Navigation */}
            <View style={styles.dateNavigation}>
              <TouchableOpacity
                onPress={() => changeDate(-1)}
                style={styles.dateArrow}
              >
                <Ionicons name="chevron-back" size={22} color="#fff" />
              </TouchableOpacity>

              <TouchableOpacity
                onPress={() => setShowDatePicker(true)}
                style={styles.dateDisplay}
              >
                <Ionicons name="calendar" size={18} color="#67BA03" />
                <Text style={styles.date}>{formatDate(selectedDate)}</Text>
              </TouchableOpacity>

              <TouchableOpacity
                onPress={() => changeDate(1)}
                style={styles.dateArrow}
              >
                <Ionicons name="chevron-forward" size={22} color="#fff" />
              </TouchableOpacity>
            </View>

            {showDatePicker && (
              <DateTimePicker
                value={selectedDate}
                mode="date"
                display="default"
                onChange={onDateChange}
              />
            )}

            <View style={styles.statsContainer}>
              <View style={styles.statBox}>
                <View style={styles.statIconContainer}>
                  <Ionicons name="checkmark-circle" size={24} color="#4CAF50" />
                </View>
                <View>
                  <Text style={styles.statNumber}>{presentCount}</Text>
                  <Text style={styles.statLabel}>Present</Text>
                </View>
              </View>
              <View style={styles.statDivider} />
              <View style={styles.statBox}>
                <View
                  style={[
                    styles.statIconContainer,
                    { backgroundColor: "#FFEBEE" },
                  ]}
                >
                  <Ionicons name="close-circle" size={24} color="#E53935" />
                </View>
                <View>
                  <Text style={[styles.statNumber, { color: "#E53935" }]}>
                    {absentCount}
                  </Text>
                  <Text style={[styles.statLabel, { color: "#E53935" }]}>
                    Absent
                  </Text>
                </View>
              </View>
            </View>
          </View>
        </LinearGradient>

        {/* Table Section */}
        <View style={styles.tableWrapper}>
          <View style={styles.tableHeaderRow}>
            <Text style={[styles.tableHeaderText, { flex: 3 }]}>
              Student Name
            </Text>
            <View style={{ flex: 0.5 }} />
            <Text
              style={[styles.tableHeaderText, { flex: 1, textAlign: "center" }]}
            >
              Status
            </Text>
            <TouchableOpacity onPress={toggleEdit} style={styles.editButton}>
              <Ionicons
                name={isEditing ? "checkmark" : "create"}
                size={16}
                color="#fff"
              />
              <Text style={styles.editButtonText}>
                {isEditing ? "Save" : "Edit"}
              </Text>
            </TouchableOpacity>
          </View>

          {/* Mark All Buttons - Only show when editing */}
          {isEditing && (
            <View style={styles.markAllContainer}>
              <TouchableOpacity
                style={styles.markAllPresentBtn}
                onPress={() => markAllStudents("Present")}
              >
                <Ionicons name="checkmark-done" size={18} color="#fff" />
                <Text style={styles.markAllText}>Mark All Present</Text>
              </TouchableOpacity>
              <TouchableOpacity
                style={styles.markAllAbsentBtn}
                onPress={() => markAllStudents("Absent")}
              >
                <Ionicons name="close" size={18} color="#fff" />
                <Text style={styles.markAllText}>Mark All Absent</Text>
              </TouchableOpacity>
            </View>
          )}

          <ScrollView style={styles.tableContainer}>
            {currentStudents.map((item, index) => (
              <View
                key={item.id}
                style={[
                  styles.tableRow,
                  index % 2 === 0 && styles.tableRowEven,
                ]}
              >
                {/* Name Column */}
                <View
                  style={{
                    flex: 3,
                    flexDirection: "row",
                    alignItems: "center",
                  }}
                >
                  <View style={styles.avatarPlaceholder}>
                    <Text style={styles.avatarText}>
                      {item.name.charAt(0).toUpperCase()}
                    </Text>
                  </View>
                  <Text style={styles.tableText}>{item.name}</Text>
                </View>

                {/* Indicator Column */}
                <View
                  style={{
                    flex: 0.5,
                    alignItems: "center",
                    justifyContent: "center",
                  }}
                >
                  {item.excuseStatus && (
                    <View
                      style={[
                        styles.excuseIndicator,
                        {
                          backgroundColor: getExcuseStatusColor(
                            item.excuseStatus
                          ),
                        },
                      ]}
                    />
                  )}
                </View>

                {/* Status Column */}
                <View style={[styles.tableCell, { flex: 1 }]}>
                  {isEditing ? (
                    <View style={styles.editButtons}>
                      <TouchableOpacity
                        style={[
                          styles.statusBtn,
                          item.status === "Present"
                            ? styles.presentSelected
                            : styles.unselectedBtn,
                        ]}
                        onPress={() => handleStatusChange(item.id, "Present")}
                      >
                        <Text
                          style={[
                            styles.btnText,
                            item.status === "Present"
                              ? { color: "#fff" }
                              : { color: "#666" },
                          ]}
                        >
                          P
                        </Text>
                      </TouchableOpacity>

                      <TouchableOpacity
                        style={[
                          styles.statusBtn,
                          item.status === "Absent"
                            ? styles.absentSelected
                            : styles.unselectedBtn,
                        ]}
                        onPress={() => handleStatusChange(item.id, "Absent")}
                      >
                        <Text
                          style={[
                            styles.btnText,
                            item.status === "Absent"
                              ? { color: "#fff" }
                              : { color: "#666" },
                          ]}
                        >
                          A
                        </Text>
                      </TouchableOpacity>
                    </View>
                  ) : (
                    <TouchableOpacity
                      disabled={item.status !== "Absent"}
                      onPress={() => openOverlay(item)}
                      style={[
                        styles.statusBadge,
                        item.status === "Present"
                          ? styles.statusBadgePresent
                          : styles.statusBadgeAbsent,
                      ]}
                    >
                      <Text
                        style={[
                          styles.statusText,
                          {
                            color:
                              item.status === "Present" ? "#4CAF50" : "#E53935",
                          },
                        ]}
                      >
                        {item.status}
                      </Text>
                    </TouchableOpacity>
                  )}
                </View>
              </View>
            ))}
          </ScrollView>

          {/* Pagination */}
          <View style={styles.pagination}>
            <TouchableOpacity
              disabled={page === 1}
              onPress={() => setPage(page - 1)}
              style={[styles.pageButton, page === 1 && styles.disabledPage]}
            >
              <Ionicons name="chevron-back" size={18} color="#fff" />
            </TouchableOpacity>
            <Text style={styles.pageText}>
              Page {page} of {totalPages}
            </Text>
            <TouchableOpacity
              disabled={page === totalPages}
              onPress={() => setPage(page + 1)}
              style={[
                styles.pageButton,
                page === totalPages && styles.disabledPage,
              ]}
            >
              <Ionicons name="chevron-forward" size={18} color="#fff" />
            </TouchableOpacity>
          </View>
        </View>

        {/* Overlay */}
        {selectedStudent && (
          <Animated.View
            style={[
              styles.overlayContainer,
              { transform: [{ translateY: slideAnim }] },
            ]}
          >
            <TouchableWithoutFeedback onPress={Keyboard.dismiss}>
              <View style={{ flex: 1 }} />
            </TouchableWithoutFeedback>

            <KeyboardAvoidingView
              behavior={Platform.OS === "ios" ? "padding" : "height"}
              style={{ maxHeight: "85%" }}
            >
              <ScrollView
                style={styles.overlayContent}
                keyboardShouldPersistTaps="handled"
              >
                <View style={styles.overlayHeader}>
                  <View>
                    <Text style={styles.overlayName}>
                      {selectedStudent.name}
                    </Text>
                    <Text style={styles.overlayDate}>
                      {formatDate(selectedDate)}
                    </Text>
                  </View>
                  <TouchableOpacity
                    onPress={closeOverlay}
                    style={styles.closeIconButton}
                  >
                    <Ionicons name="close-circle" size={28} color="#999" />
                  </TouchableOpacity>
                </View>

                <Text style={styles.excuseHeader}>Excuse Letter Review</Text>

                <View style={styles.excuseCard}>
                  <View style={styles.excuseRow}>
                    <View style={styles.excuseIconBox}>
                      <Ionicons
                        name="document-text"
                        size={20}
                        color="#67BA03"
                      />
                    </View>
                    <View style={styles.excuseRowContent}>
                      <Text style={styles.excuseLabel}>Submission Status</Text>
                      <Text
                        style={[
                          styles.excuseValue,
                          excuseLetter?.status === "Pending" &&
                            styles.statusPending,
                          excuseLetter?.status === "Approved" &&
                            styles.statusApproved,
                          excuseLetter?.status === "Denied" &&
                            styles.statusDenied,
                        ]}
                      >
                        {excuseLetter?.status || "No submission yet"}
                      </Text>
                    </View>
                  </View>

                  <View style={styles.excuseDivider} />

                  <View style={styles.excuseRow}>
                    <View style={styles.excuseIconBox}>
                      <Ionicons name="time" size={20} color="#67BA03" />
                    </View>
                    <View style={styles.excuseRowContent}>
                      <Text style={styles.excuseLabel}>Submitted On</Text>
                      <Text style={styles.excuseValue}>
                        {excuseLetter?.time
                          ? new Date(excuseLetter.time).toLocaleString()
                          : "—"}
                      </Text>
                    </View>
                  </View>

                  <View style={styles.excuseDivider} />

                  <View style={styles.excuseRow}>
                    <View style={styles.excuseIconBox}>
                      <Ionicons name="attach" size={20} color="#67BA03" />
                    </View>
                    <View style={styles.excuseRowContent}>
                      <Text style={styles.excuseLabel}>File Attachment</Text>
                      {excuseLetter?.fileName ? (
                        <TouchableOpacity
                          onPress={openPdfInApp}
                          style={styles.fileButton}
                        >
                          <Ionicons
                            name="open-outline"
                            size={14}
                            color="#67BA03"
                          />
                        </TouchableOpacity>
                      ) : (
                        <Text style={styles.excuseValue}>No file attached</Text>
                      )}
                    </View>
                  </View>

                  <View style={styles.excuseDivider} />

                  <View style={styles.excuseRow}>
                    <View style={styles.excuseIconBox}>
                      <Ionicons name="chatbox" size={20} color="#67BA03" />
                    </View>
                    <View style={styles.excuseRowContent}>
                      <Text style={styles.excuseLabel}>Teacher Comments</Text>
                      <TextInput
                        style={styles.commentInput}
                        placeholder="Add your comments here..."
                        placeholderTextColor="#999"
                        value={comment}
                        onChangeText={setComment}
                        multiline
                      />
                    </View>
                  </View>
                </View>

                {/* PDF Viewer */}
                {showPdf && excuseLetter?.file && (
                  <View style={styles.pdfViewerContainer}>
                    <View style={styles.pdfHeader}>
                      <Text style={styles.pdfTitle}>
                        {excuseLetter.fileName}
                      </Text>
                      <TouchableOpacity onPress={() => setShowPdf(false)}>
                        <Text style={styles.closePdfText}>Close PDF</Text>
                      </TouchableOpacity>
                    </View>
                    <WebView
                      originWhitelist={["*"]}
                      source={{
                        uri: `data:application/pdf;base64,${excuseLetter.file}`,
                      }}
                      style={styles.webView}
                    />
                  </View>
                )}

                {/* Action Buttons */}
                <View style={styles.overlayButtons}>
                  <TouchableOpacity
                    style={styles.approveBtn}
                    onPress={() =>
                      Alert.alert(
                        "Approve Submission",
                        "Approve this excuse letter?",
                        [
                          { text: "Cancel", style: "cancel" },
                          {
                            text: "Approve",
                            onPress: () => updateExcuseStatus("Approved"),
                          },
                        ]
                      )
                    }
                  >
                    <Ionicons name="checkmark-circle" size={20} color="#fff" />
                    <Text style={styles.btnActionText}>APPROVE</Text>
                  </TouchableOpacity>
                  <TouchableOpacity
                    style={styles.denyBtn}
                    onPress={() =>
                      Alert.alert(
                        "Deny Submission",
                        "Deny this excuse letter?",
                        [
                          { text: "Cancel", style: "cancel" },
                          {
                            text: "Deny",
                            onPress: () => updateExcuseStatus("Denied"),
                          },
                        ]
                      )
                    }
                  >
                    <Ionicons name="close-circle" size={20} color="#fff" />
                    <Text style={styles.btnActionText}>DENY</Text>
                  </TouchableOpacity>
                </View>
              </ScrollView>
            </KeyboardAvoidingView>
          </Animated.View>
        )}
      </SafeAreaView>
    </>
  );
}
