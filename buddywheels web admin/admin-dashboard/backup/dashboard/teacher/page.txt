"use client";

import { useEffect, useState } from "react";
import { useRouter } from "next/navigation";
import { onAuthStateChanged, signOut } from "firebase/auth";
import {
  collection,
  getDocs,
  query,
  where,
  deleteDoc,
  doc,
  updateDoc,
} from "firebase/firestore";
import { auth, db } from "@/lib/firebaseConfig";
import { motion, AnimatePresence } from "framer-motion";
import { X, Edit2 } from "lucide-react";

interface Teacher {
  id: string;
  fullName: string;
  email: string;
  phone: string;
  idNumber: string;
  ppic?: string;
}

export default function TeacherPage() {
  const router = useRouter();
  const [teachers, setTeachers] = useState<Teacher[]>([]);
  const [loading, setLoading] = useState(true);
  const [editMode, setEditMode] = useState(false);
  const [selectedTeachers, setSelectedTeachers] = useState<string[]>([]);
  const [editingTeacher, setEditingTeacher] = useState<Teacher | null>(null);
  const [showEditModal, setShowEditModal] = useState(false);

  const DEFAULT_PROFILE_PIC =
    "https://cdn-icons-png.flaticon.com/512/3135/3135715.png";

  // 🔹 Helper to safely render Base64 or URL images
  const getSafeImageSrc = (img?: string) => {
    const fallback = DEFAULT_PROFILE_PIC;

    if (!img) return fallback;

    // If it's a valid Base64 with data URI
    if (/^data:image\/(png|jpeg|jpg|gif);base64,[A-Za-z0-9+/=]+$/.test(img)) {
      return img;
    }

    // If it's raw Base64 without data URI prefix
    if (/^[A-Za-z0-9+/=]+$/.test(img)) {
      return `data:image/png;base64,${img}`;
    }

    // Otherwise assume it's a valid URL
    return img;
  };

  // 🔹 Auth check
  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, (user) => {
      if (!user || user.email !== process.env.NEXT_PUBLIC_ADMIN_EMAIL) {
        router.push("/");
      }
    });
    return () => unsubscribe();
  }, [router]);

  // 🔹 Fetch teachers + profile pics
  const fetchTeachers = async () => {
    setLoading(true);
    try {
      const teachersRef = collection(db, "Teacher");
      const snapshot = await getDocs(teachersRef);

      const teacherList = await Promise.all(
        snapshot.docs.map(async (docSnap) => {
          const data = docSnap.data();

          // Query users collection where 'id' field equals teacher's id
          const usersQuery = query(
            collection(db, "users"),
            where("id", "==", data.id)
          );
          const userQuerySnapshot = await getDocs(usersQuery);

          let ppic = "";
          if (!userQuerySnapshot.empty) {
            const userData = userQuerySnapshot.docs[0].data();
            ppic = userData.ppic || "";
          }

          return { ...data, ppic } as Teacher;
        })
      );

      setTeachers(teacherList);
    } catch (error) {
      console.error("Error fetching teachers:", error);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchTeachers();
  }, []);

  const handleLogout = async () => {
    await signOut(auth);
    router.push("/");
  };

  // ✅ Toggle teacher selection for deletion
  const toggleSelectTeacher = (teacherID: string) => {
    setSelectedTeachers((prev) =>
      prev.includes(teacherID)
        ? prev.filter((id) => id !== teacherID)
        : [...prev, teacherID]
    );
  };

  // ✅ Delete selected teachers
  const handleDeleteTeachers = async () => {
    if (selectedTeachers.length === 0) return alert("No teachers selected.");
    const confirmed = confirm(
      "⚠️ Are you sure you want to delete these teachers?"
    );
    if (!confirmed) return;

    try {
      for (const id of selectedTeachers) {
        await deleteDoc(doc(db, "Teacher", id));
      }
      alert("🗑️ Selected teachers deleted successfully.");
      setSelectedTeachers([]);
      setEditMode(false);
      fetchTeachers();
    } catch (error) {
      console.error("Error deleting teachers:", error);
      alert("Error deleting some teachers.");
    }
  };

  // ✅ Edit individual teacher
  const handleEditTeacher = (teacher: Teacher) => {
    setEditingTeacher({ ...teacher });
    setShowEditModal(true);
  };

  const handleSaveEdit = async () => {
    if (!editingTeacher) return;

    try {
      const teacherRef = doc(db, "Teacher", editingTeacher.id);
      await updateDoc(teacherRef, {
        fullName: editingTeacher.fullName,
        email: editingTeacher.email,
        phone: editingTeacher.phone,
        idNumber: editingTeacher.idNumber,
      });

      // Also update ppic in users collection if needed
      const usersQuery = query(
        collection(db, "users"),
        where("id", "==", editingTeacher.id)
      );
      const userQuerySnapshot = await getDocs(usersQuery);

      if (!userQuerySnapshot.empty) {
        const userDocRef = userQuerySnapshot.docs[0].ref;
        await updateDoc(userDocRef, {
          ppic: editingTeacher.ppic || "",
        });
      }

      alert("✅ Teacher updated successfully!");
      setShowEditModal(false);
      setEditingTeacher(null);
      fetchTeachers();
    } catch (error) {
      console.error("Error updating teacher:", error);
      alert("Error updating teacher.");
    }
  };

  return (
    <div className="min-h-screen bg-linear-to-br from-green-100 via-white to-green-50">
      {/* Header */}
      <header className="flex justify-between items-center px-8 py-4 bg-white shadow-md sticky top-0 z-40">
        <div className="flex items-center space-x-3">
          <img
            src="/logo.png"
            alt="BuddyWheels Logo"
            className="w-30 h-10 cursor-pointer"
            onClick={() => router.push("/dashboard")}
          />
          <h1 className="text-xl font-semibold text-gray-700">Teachers</h1>
        </div>
        <div className="flex items-center space-x-3">
          <button
            onClick={() => router.push("/dashboard")}
            className="px-4 py-2 bg-gray-200 rounded-lg hover:bg-gray-300 transition text-gray-700"
          >
            ⬅ Back to Dashboard
          </button>
          <button
            onClick={handleLogout}
            className="bg-green-500 text-white px-4 py-2 rounded-lg hover:bg-green-600 active:scale-95 transition"
          >
            Logout
          </button>
        </div>
      </header>

      {/* Main Section */}
      <main className="p-10">
        <motion.div
          initial={{ opacity: 0, y: 30 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.6 }}
          className="text-center mb-10"
        >
          <h2 className="text-3xl font-bold text-green-700">Teachers 👩‍🏫</h2>
          <p className="text-gray-600 mt-2">
            List of all teachers in the system.
          </p>
        </motion.div>

        {loading ? (
          <p className="text-center text-gray-500">Loading teachers...</p>
        ) : teachers.length === 0 ? (
          <p className="text-center text-gray-500 italic">No teachers found.</p>
        ) : (
          <div className="bg-white rounded-xl shadow-lg p-6">
            {/* Header Row with Edit/Delete Buttons */}
            <div className="flex justify-between items-center mb-6">
              <div className="flex items-center space-x-3">
                <h3 className="text-xl font-semibold text-green-700">
                  Teacher List
                </h3>
                <span className="bg-green-100 text-green-700 text-sm px-3 py-1 rounded-full">
                  Total: {teachers.length}
                </span>
              </div>

              <div className="flex space-x-3">
                {editMode && (
                  <button
                    onClick={handleDeleteTeachers}
                    className="bg-red-500 text-white px-4 py-2 rounded-lg hover:bg-red-600 active:scale-95 transition"
                  >
                    🗑 Delete Selected
                  </button>
                )}
                <button
                  onClick={() => {
                    setEditMode(!editMode);
                    setSelectedTeachers([]);
                  }}
                  className="bg-yellow-500 text-white px-4 py-2 rounded-lg hover:bg-yellow-600 active:scale-95 transition"
                >
                  {editMode ? "Cancel Edit" : "✏️ Edit"}
                </button>
              </div>
            </div>

            {/* Table */}
            <div className="overflow-x-auto">
              <table className="w-full border text-left text-gray-600 border-gray-500">
                <thead className="bg-green-100">
                  <tr>
                    <th className="py-2 px-3 text-gray-700 font-medium border-b">
                      Profile
                    </th>
                    <th className="py-2 px-3 text-gray-700 font-medium border-b">
                      ID
                    </th>
                    <th className="py-2 px-3 text-gray-700 font-medium border-b">
                      Full Name
                    </th>
                    <th className="py-2 px-3 text-gray-700 font-medium border-b">
                      Email
                    </th>
                    <th className="py-2 px-3 text-gray-700 font-medium border-b">
                      Phone
                    </th>
                    <th className="py-2 px-3 text-gray-700 font-medium border-b">
                      IC Number
                    </th>
                    {editMode && (
                      <>
                        <th className="py-2 px-3 text-gray-700 font-medium border-b text-center">
                          Edit
                        </th>
                        <th className="py-2 px-3 text-gray-700 font-medium border-b text-center">
                          Select
                        </th>
                      </>
                    )}
                  </tr>
                </thead>
                <tbody>
                  {teachers.map((teacher) => (
                    <tr
                      key={teacher.id}
                      className="hover:bg-green-50 transition"
                    >
                      <td className="py-2 px-3 border-b border-gray-500">
                        <img
                          src={getSafeImageSrc(teacher.ppic)}
                          alt={teacher.fullName}
                          className="w-10 h-10 rounded-full object-cover"
                          onError={(e) => {
                            (e.target as HTMLImageElement).src =
                              DEFAULT_PROFILE_PIC;
                          }}
                        />
                      </td>
                      <td className="py-2 px-3 border-b text-gray-600 border-gray-500">
                        {teacher.id}
                      </td>
                      <td className="py-2 px-3 border-b text-gray-600 border-gray-500">
                        {teacher.fullName}
                      </td>
                      <td className="py-2 px-3 border-b text-gray-600 border-gray-500">
                        {teacher.email}
                      </td>
                      <td className="py-2 px-3 border-b text-gray-600 border-gray-500">
                        {teacher.phone}
                      </td>
                      <td className="py-2 px-3 border-b text-gray-600 border-gray-500">
                        {teacher.idNumber}
                      </td>
                      {editMode && (
                        <>
                          <td className="py-2 px-3 border-b border-gray-500 text-center">
                            <button
                              onClick={() => handleEditTeacher(teacher)}
                              className="text-blue-500 hover:text-blue-700 transition"
                            >
                              <Edit2 className="w-5 h-5" />
                            </button>
                          </td>
                          <td className="py-2 px-3 border-b border-gray-500 text-center">
                            <input
                              type="checkbox"
                              checked={selectedTeachers.includes(teacher.id)}
                              onChange={() => toggleSelectTeacher(teacher.id)}
                            />
                          </td>
                        </>
                      )}
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* Footer */}
        <footer className="text-center text-gray-500 mt-16">
          <p>
            © {new Date().getFullYear()} BuddyWheels Admin Panel. All rights
            reserved.
          </p>
        </footer>
      </main>

      {/* Edit Teacher Modal */}
      <AnimatePresence>
        {showEditModal && editingTeacher && (
          <motion.div
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            className="fixed inset-0 bg-black bg-opacity-40 flex justify-center items-center z-50"
          >
            <motion.div
              initial={{ scale: 0.9, opacity: 0 }}
              animate={{ scale: 1, opacity: 1 }}
              exit={{ scale: 0.9, opacity: 0 }}
              className="bg-white rounded-2xl p-8 shadow-xl w-[90%] max-w-lg relative max-h-[90vh] overflow-y-auto"
            >
              <button
                onClick={() => {
                  setShowEditModal(false);
                  setEditingTeacher(null);
                }}
                className="absolute top-4 right-4 text-gray-500 hover:text-gray-700"
              >
                <X className="w-6 h-6" />
              </button>

              <h3 className="text-xl font-semibold text-gray-700 mb-6">
                Edit Teacher Details
              </h3>

              <div className="space-y-4">
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-1">
                    Teacher ID (Read-only)
                  </label>
                  <input
                    type="text"
                    value={editingTeacher.id}
                    disabled
                    className="w-full border rounded-lg p-2 bg-gray-100 text-gray-500"
                  />
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-1">
                    Profile Picture URL
                  </label>
                  <div className="flex items-center gap-3">
                    <img
                      src={getSafeImageSrc(editingTeacher.ppic)}
                      alt="Preview"
                      className="w-12 h-12 rounded-full object-cover border-2 border-gray-200"
                      onError={(e) => {
                        (e.target as HTMLImageElement).src =
                          DEFAULT_PROFILE_PIC;
                      }}
                    />
                    <input
                      type="text"
                      value={editingTeacher.ppic || ""}
                      onChange={(e) =>
                        setEditingTeacher({
                          ...editingTeacher,
                          ppic: e.target.value,
                        })
                      }
                      placeholder="Enter image URL or leave empty for default"
                      className="flex-1 border rounded-lg p-2 placeholder-gray-400 text-gray-500"
                    />
                  </div>
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-1">
                    Full Name *
                  </label>
                  <input
                    type="text"
                    value={editingTeacher.fullName}
                    onChange={(e) =>
                      setEditingTeacher({
                        ...editingTeacher,
                        fullName: e.target.value,
                      })
                    }
                    className="w-full border rounded-lg p-2 placeholder-gray-500 text-gray-500"
                  />
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-1">
                    Email *
                  </label>
                  <input
                    type="email"
                    value={editingTeacher.email}
                    onChange={(e) =>
                      setEditingTeacher({
                        ...editingTeacher,
                        email: e.target.value,
                      })
                    }
                    className="w-full border rounded-lg p-2 placeholder-gray-500 text-gray-500"
                  />
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-1">
                    Phone *
                  </label>
                  <input
                    type="text"
                    value={editingTeacher.phone}
                    onChange={(e) =>
                      setEditingTeacher({
                        ...editingTeacher,
                        phone: e.target.value,
                      })
                    }
                    className="w-full border rounded-lg p-2 placeholder-gray-500 text-gray-500"
                  />
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-1">
                    IC Number *
                  </label>
                  <input
                    type="text"
                    value={editingTeacher.idNumber}
                    onChange={(e) =>
                      setEditingTeacher({
                        ...editingTeacher,
                        idNumber: e.target.value,
                      })
                    }
                    className="w-full border rounded-lg p-2 placeholder-gray-500 text-gray-500"
                  />
                </div>
              </div>

              <div className="flex justify-end mt-6 space-x-3">
                <button
                  onClick={() => {
                    setShowEditModal(false);
                    setEditingTeacher(null);
                  }}
                  className="px-4 py-2 bg-gray-200 rounded-lg hover:bg-gray-300 transition text-gray-700"
                >
                  Cancel
                </button>
                <button
                  onClick={handleSaveEdit}
                  className="px-4 py-2 bg-blue-500 text-white rounded-lg hover:bg-blue-600 transition"
                >
                  Save Changes
                </button>
              </div>
            </motion.div>
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
}
